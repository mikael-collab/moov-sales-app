const SID = '147Zig2OPV_daJN8i6U_EwANeTYVInWymh6qwq9C27uc'; function colLetter(n) { if (n < 26) return String.fromCharCode(65 + n); return String.fromCharCode(64 + Math.floor(n / 26)) + String.fromCharCode(65 + (n % 26)); } function corsOutput(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { try { const p = e.parameter || {}; const sheetName = p.sheet || ''; const limit = p.limit ? parseInt(p.limit) : 0; const ss = SpreadsheetApp.openById(SID); // No sheet param → return list of all sheet names (used for debugging) if (!sheetName) { const names = ss.getSheets().map(function(s) { return s.getName(); }); return corsOutput({ ok: true, sheets: names }); } const sheet = ss.getSheetByName(sheetName); if (!sheet) { const names = ss.getSheets().map(function(s) { return s.getName(); }); return corsOutput({ error: 'Sheet not found: [' + sheetName + ']', available: names }); } const range = sheet.getDataRange(); const allValues = range.getDisplayValues(); // preserves leading zeros & date formats const allBgs = range.getBackgrounds(); // actual cell background colors per row if (allValues.length < 2) return corsOutput({ data: [] }); let rows = allValues.slice(1); let bgRows = allBgs.slice(1); if (limit > 0) { rows = rows.slice(-limit); bgRows = bgRows.slice(-limit); } const result = rows.map(function(row, ri) { const obj = {}; row.forEach(function(cell, ci) { obj[colLetter(ci)] = cell ? cell.toString().trim() : ''; }); // Use column A background as the row status color // Fall back to white if null const rawBg = bgRows[ri][0] || '#ffffff'; obj._bg = rawBg.toLowerCase() === '#ffffff' || rawBg.toLowerCase() === '#fff' || rawBg === 'white' ? '#ffffff' : rawBg; return obj; }); return corsOutput({ data: result }); } catch (err) { return corsOutput({ error: err.message }); } } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SID); if (payload.action === 'addCustomer') { const sheet = ss.getSheetByName('לקוחות'); if (!sheet) return corsOutput({ error: 'לקוחות sheet not found' }); const keys = ['A','B','C','D','E','F','G','H','I','J','K']; const row = keys.map(function(k) { return payload.data[k] || ''; }); const nextRow = sheet.getLastRow() + 1; const range = sheet.getRange(nextRow, 1, 1, row.length); range.setNumberFormat('@'); // force text → preserves leading zeros range.setValues([row]); return corsOutput({ status: 'ok', savedRow: nextRow }); } return corsOutput({ error: 'Unknown action' }); } catch (err) { return corsOutput({ error: err.message }); } } // ── Run manually to check sheet names and row counts ── function testConnection() { const ss = SpreadsheetApp.openById(SID); const sheets = ss.getSheets(); sheets.forEach(function(s) { Logger.log('[' + s.getName() + '] → ' + (s.getLastRow() - 1) + ' data rows'); }); }
